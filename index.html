// ─── ЦБА / CBCA — 19 критериев в 5 блоках ───

// ─── Рендер ───
const box = document.getElementById('criteria');
let idx = 0;
CBCA.forEach(g => {
  const h = document.createElement('h2');
  h.textContent = g.block;
  box.appendChild(h);

  const card = document.createElement('div');
  card.className = 'card';

  g.items.forEach(([title, desc]) => {
    idx++;
    const l = document.createElement('label');
    l.className = 'crit';
    l.innerHTML = `
      <input type="checkbox" data-i="${idx-1}">
      <div>
        <div class="title"><span class="num">${idx}.</span>${title}</div>
        <div class="desc">${desc}</div>
      </div>`;
    card.appendChild(l);
  });
  box.appendChild(card);
});

const checks  = () => [...document.querySelectorAll('input[type=checkbox]')];
const countEl = document.getElementById('count');
const fillEl  = document.getElementById('fill');
const verdict = document.getElementById('verdict');

function update() {
  const n = checks().filter(c => c.checked).length;
  countEl.textContent = n;
  fillEl.style.width = (n / 19 * 100) + '%';

  verdict.classList.remove('ok', 'warn');
  let text;
  if (n <= 5) {
    verdict.classList.add('warn');
    text = "Мало признаков правдоподобия. Показания выглядят шаблонно или заученно.";
  } else if (n <= 10) {
    text = "Смешанная картина. Отдельные признаки живого опыта есть, но их недостаточно для уверенного вывода.";
  } else if (n <= 14) {
    verdict.classList.add('ok');
    text = "Умеренно правдоподобно. Присутствует значительная часть критериев живой памяти.";
  } else {
    verdict.classList.add('ok');
    text = "Высокое правдоподобие. Показания насыщены признаками личного опыта.";
  }
  verdict.textContent = `Присутствует ${n} из 19 критериев. ${text}`;
}

document.body.addEventListener('change', e => {
  if (e.target.type === 'checkbox') update();
});

document.getElementById('reset').onclick = () => {
  checks().forEach(c => c.checked = false);
  document.getElementById('subject').value = '';
  update();
};

document.getElementById('copy').onclick = () => {
  const name = document.getElementById('subject').value.trim() || '(без названия)';
  let out = `Детектор лжи — отчёт ЦБА\nОбъект: ${name}\nДата: ${new Date().toLocaleString()}\n\n`;
  let i = 0;
  CBCA.forEach(g => {
    out += `${g.block}\n`;
    g.items.forEach(([t]) => {
      out += `${checks()[i].checked ? '[x]' : '[ ]'} ${i+1}. ${t}\n`;
      i++;
    });
    out += '\n';
  });
  const n = checks().filter(c => c.checked).length;
  out += `Итог: ${n}/19\n${verdict.textContent}`;

  navigator.clipboard.writeText(out).then(
    () => alert('Отчёт скопирован в буфер обмена'),
    () => alert('Не удалось скопировать.')
  );
};

document.getElementById('save').onclick = () => {
  const data = {
    subject: document.getElementById('subject').value,
    checked: checks().map(c => c.checked)
  };
  localStorage.setItem('cbcaSave', JSON.stringify(data));
  alert('Сохранено в браузере');
};

(function load() {
  try {
    const raw = localStorage.getItem('cbcaSave');
    if (!raw) return update();
    const d = JSON.parse(raw);
    document.getElementById('subject').value = d.subject || '';
    (d.checked || []).forEach((v, i) => { if (checks()[i]) checks()[i].checked = v; });
  } catch(e) {}
  update();
})();
